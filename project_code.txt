#include <ESP8266WiFi.h>
#include <WiFiClient.h>
#include <ThingSpeak.h> // IoT Platform
#include <Ticker.h>

// WiFi Credentials
const char* ssid = "YOUR_SSID";
const char* password = "YOUR_PASSWORD";

// ThingSpeak Info
unsigned long myChannelNumber = YOUR_CHANNEL_NUMBER;
const char* myWriteAPIKey = "YOUR_API_KEY";

// Define Pins
const int pulsePin = A0;        // Pulse sensor
const int electrolytePin = A0;  // Same pin; in real case use multiplexer or second ADC
const int emergencyButton = D1;
const int stepTimerButton = D2;

// Variables
WiFiClient client;
Ticker timer;
bool timerRunning = false;
int steps = 0;

// Mock function for reading heart rate
int readHeartRate() {
  int sensorValue = analogRead(pulsePin);
  // Simulate BPM range
  return map(sensorValue, 0, 1023, 60, 120);
}

// Mock function for reading electrolyte levels
float readElectrolyteLevel() {
  int analogVal = analogRead(electrolytePin);
  return analogVal * (3.3 / 1023.0); // Simulated voltage level
}

void emergencyAlert() {
  Serial.println("Emergency button pressed!");
  int bpm = readHeartRate();
  float electrolyte = readElectrolyteLevel();

  ThingSpeak.setField(1, bpm);
  ThingSpeak.setField(2, electrolyte);
  ThingSpeak.setField(3, steps);
  ThingSpeak.setStatus("EMERGENCY ALERT!");
  ThingSpeak.writeFields(myChannelNumber, myWriteAPIKey);
}

void stepTimerToggle() {
  steps += 1;
  timerRunning = !timerRunning;
  Serial.print("Steps: ");
  Serial.println(steps);
  Serial.print("Timer running: ");
  Serial.println(timerRunning ? "Yes" : "No");
}

void sendData() {
  int bpm = readHeartRate();
  float electrolyte = readElectrolyteLevel();

  Serial.println("Sending data...");
  Serial.print("BPM: "); Serial.println(bpm);
  Serial.print("Electrolyte (V): "); Serial.println(electrolyte);

  ThingSpeak.setField(1, bpm);
  ThingSpeak.setField(2, electrolyte);
  ThingSpeak.setField(3, steps);
  ThingSpeak.setStatus("Normal");
  ThingSpeak.writeFields(myChannelNumber, myWriteAPIKey);
}

void setup() {
  Serial.begin(115200);
  pinMode(emergencyButton, INPUT_PULLUP);
  pinMode(stepTimerButton, INPUT_PULLUP);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Connecting to WiFi...");
  }
  Serial.println("Connected.");

  ThingSpeak.begin(client);

  // Attach button interrupts
  attachInterrupt(digitalPinToInterrupt(emergencyButton), emergencyAlert, FALLING);
  attachInterrupt(digitalPinToInterrupt(stepTimerButton), stepTimerToggle, FALLING);

  // Start periodic sending
  timer.attach(30, sendData); // send data every 30 seconds
}

void loop() {
  // Nothing to do in loop, interrupt and timer handle tasks
}
